cmake_minimum_required(VERSION 3.24)
project(EncryptedResourceCase LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This independent case requires native Windows and an MSVC-compatible compiler")
endif()
if(CMAKE_CONFIGURATION_TYPES OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use an x64 single-configuration generator (Ninja or JOM)")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(AR_ENCRYPT_RESOURCES "Embed authenticated encrypted resources" OFF)
set(AR_RESOURCE_KEY_FILE "" CACHE FILEPATH "Private resource key/nonce state; generate explicitly with resource-packer keygen")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(resource-support STATIC support.cpp)
target_compile_definitions(resource-support PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(resource-support PUBLIC bcrypt)
add_executable(resource-packer packer.cpp)
target_link_libraries(resource-packer PRIVATE resource-support)
add_executable(resource-demo consumer.cpp)
target_link_libraries(resource-demo PRIVATE resource-support)
foreach(target resource-support resource-packer resource-demo)
    target_compile_options(${target} PRIVATE /W4 /EHsc)
endforeach()
foreach(target resource-packer resource-demo)
    target_link_options(${target} PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO
        "/PDBALTPATH:$<TARGET_PDB_FILE_NAME:${target}>")
endforeach()
file(GLOB_RECURSE resources CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/resources"
    "${CMAKE_CURRENT_SOURCE_DIR}/resources/*")
list(SORT resources)
string(JOIN "\n" inventory ${resources})
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/inventory.txt" CONTENT "${inventory}\n" @ONLY)
set(resource_inputs)
foreach(resource IN LISTS resources)
    list(APPEND resource_inputs "${CMAKE_CURRENT_SOURCE_DIR}/resources/${resource}")
endforeach()
set(identity "resource-case-v1;${CMAKE_CXX_COMPILER_ID};${CMAKE_CXX_COMPILER_VERSION};${CMAKE_SIZEOF_VOID_P};${CMAKE_BUILD_TYPE};${CMAKE_CXX_FLAGS};${CMAKE_CXX_FLAGS_DEBUG};${CMAKE_CXX_FLAGS_RELEASE}")
foreach(input consumer.cpp support.cpp support.hpp "${CMAKE_CXX_COMPILER}")
    if(NOT IS_ABSOLUTE "${input}")
        set(input "${CMAKE_CURRENT_SOURCE_DIR}/${input}")
    endif()
    file(SHA256 "${input}" fingerprint)
    string(APPEND identity ";${fingerprint}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input}")
endforeach()
string(SHA256 identity "${identity}")
target_compile_definitions(resource-demo PRIVATE AR_PRODUCER_ID="${identity}"
    AR_ENCRYPT_RESOURCES=$<BOOL:${AR_ENCRYPT_RESOURCES}>)
if(AR_ENCRYPT_RESOURCES)
    if(NOT EXISTS "${AR_RESOURCE_KEY_FILE}")
        message(FATAL_ERROR "AR_RESOURCE_KEY_FILE must select an explicitly generated private key")
    endif()
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/private")
    add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/private/resource_bundle.hpp"
        "${CMAKE_BINARY_DIR}/private/resources.arc"
        COMMAND resource-packer pack "${CMAKE_CURRENT_SOURCE_DIR}/resources"
            "${CMAKE_BINARY_DIR}/inventory.txt" "${AR_RESOURCE_KEY_FILE}"
            "${CMAKE_BINARY_DIR}/private/resource_bundle.hpp" "${CMAKE_BINARY_DIR}/private/resources.arc"
        DEPENDS resource-packer ${resource_inputs} "${CMAKE_BINARY_DIR}/inventory.txt" "${AR_RESOURCE_KEY_FILE}"
        VERBATIM)
    target_sources(resource-demo PRIVATE "${CMAKE_BINARY_DIR}/private/resource_bundle.hpp")
    target_include_directories(resource-demo PRIVATE "${CMAKE_BINARY_DIR}/private")
else()
    install(DIRECTORY resources DESTINATION bin)
endif()
install(TARGETS resource-demo RUNTIME DESTINATION bin)
